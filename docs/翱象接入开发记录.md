```powershell
.\.venv\Scripts\python.exe tools/verify_aoxiang_baseline.py --qa qa/aoxiang_integration_20261101 --report qa/aoxiang_integration_20261101/recheck-AX01.json
$env:LOCALAPPDATA = "$PWD/qa/aoxiang_integration_20261101/node-recheck-settings"
Push-Location components/aoxiang
try { .\runtime\node.exe --test 'test/*.test.mjs' } finally { Pop-Location }
```
